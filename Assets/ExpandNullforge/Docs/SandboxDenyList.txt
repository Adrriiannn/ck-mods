# Types the C# compiler writes into every assembly by itself, which no source file mentions and
# which the game's own checker never walks. Only DimensionSandboxAssemblyGuard uses these, because
# only it reads the built DLL, where they show up. Each was found by reading the metadata of a clean
# build of ExpandNullforge.dll and then locating where it is actually used; the note says where.
# Anything NOT listed here is reported, so a new one fails the build and a person decides.
#
# Cecil reports a modifier's ELEMENT type, not the modifier, so a modreq never reaches the checker;
# and CustomAttribute appears nowhere in Trivial.CodeSecurity, so an attribute never does either.
compiler-emitted: System.Diagnostics.DebuggableAttribute            # assembly-level [Debuggable]
compiler-emitted: System.Diagnostics.DebuggableAttribute+DebuggingModes  # its ctor's parameter type
compiler-emitted: System.Diagnostics.DebuggerBrowsableAttribute     # on auto-property backing fields
compiler-emitted: System.Diagnostics.DebuggerBrowsableState         # its ctor's parameter type
compiler-emitted: System.Diagnostics.DebuggerHiddenAttribute        # on compiler-generated members
compiler-emitted: System.Runtime.InteropServices.InAttribute        # modreq on every "in" parameter
compiler-emitted: System.Runtime.InteropServices.UnmanagedType      # modreq for "where T : unmanaged"
#
# These two are not the C# compiler but Burst's IL post-processor, which rewrites every
# [BurstCompile] method into a nested $BurstDirectCall type carrying a function-pointer delegate.
# They are in the assembly Unity builds and in no source file. Measured in
# Library/ScriptAssemblies/ExpandNullforge.dll: the first is an attribute type used by 8 custom
# attributes, the second appears only inside that attribute's constructor signature.
compiler-emitted: System.Runtime.InteropServices.UnmanagedFunctionPointerAttribute
compiler-emitted: System.Runtime.InteropServices.CallingConvention
